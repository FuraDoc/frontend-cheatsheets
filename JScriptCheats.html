Основные конструкции и методы для быстрого старта

1. Переменные и типы данных

let x = 5;      // Можно менять
const y = 10;     // Константа (нельзя менять)
var z = 2;       // Устаревший способ

// Примитивные типы
typeof "Текст";    // string
typeof 42;         // number
typeof true;       // boolean
typeof undefined;  // undefined
typeof null;       // object (особенность JS)
typeof 42n;        // bigint
typeof Symbol();   // symbol

// Ссылочные типы
typeof [];         // object
typeof {};         // object
typeof function(){}; // function
Важно: let и const имеют блочную область видимости, var — функциональную.

2. Функции

Function Declaration:

function sum(a, b) {
    return a + b;
}

Function Expression:

const multiply = function(a, b) {
    return a * b;
};

Стрелочные функции:

const divide = (a, b) => a / b;
const greet = name => `Hello, ${name}`;
const sayHi = () => console.log('Hi!');
const multiLine = (x, y) => {
    let result = x + y;
    return result;
};
Отличие: Стрелочные функции не имеют своего this.

3. Управляющие конструкции

Условия:

if (age >= 18) {
    console.log('Взрослый');
} else if (age >= 13) {
    console.log('Подросток');
} else {
    console.log('Ребенок');
}

// Тернарный оператор
let status = age >= 18 ? 'Взрослый' : 'Ребенок';

Циклы:

// for
for (let i = 0; i < 5; i++) {
    console.log(i);
}

// while
while (condition) {
    // код
}

// do...while
do {
    // код
} while (condition);

// for...of (для массивов)
for (let item of array) {
    console.log(item);
}

// for...in (для объектов)
for (let key in object) {
    console.log(key, object[key]);
}

4. Массивы (Array)

// Создание
let arr = [1, 2, 'текст'];

// Основные методы
arr.push('новый');          // Добавить в конец
arr.pop();                    // Удалить с конца
arr.unshift('начало');       // Добавить в начало
arr.shift();                  // Удалить с начала
arr.indexOf('текст');       // Найти индекс
arr.includes(2);            // Проверить наличие
arr.slice(1, 3);           // Копировать часть
arr.splice(1, 1, 'новый');  // Удалить/вставить
arr.concat([4, 5]);        // Объединить
arr.join('-');              // В строку "1-2-текст"
arr.reverse();               // Перевернуть
arr.sort();                  // Отсортировать
arr.length                   // Длина массива

// Перебор
arr.forEach(function(item) {
    console.log(item);
});

// Поиск
arr.find(item => item > 1);      // Первый элемент
arr.filter(item => item > 1);     // Все подходящие
arr.map(item => item * 2);        // Преобразовать
arr.reduce((sum, item) => sum + item, 0); // Сумма

5. Объекты (Object)

// Создание
let user = {
    name: 'Анна',
    age: 25,
    'любимый цвет': 'синий',  // Ключ с пробелом
    sayHi: function() {
        console.log(`Привет, ${this.name}`);
    }
};

// Доступ
user.name;                     // 'Анна'
user['любимый цвет'];        // 'синий'
user.sayHi();                  // 'Привет, Анна'

// Изменение
user.age = 26;
user.city = 'Москва';         // Добавление

// Удаление
delete user.city;

// Проверка наличия
'name' in user;             // true
user.hasOwnProperty('name');   // true

// Перебор свойств
for (let key in user) {
    console.log(key, user[key]);
}

// Деструктуризация
let {name, age} = user;
console.log(name, age);        // 'Анна' 26

6. Работа со строками (String)

let str = "Hello, World!";

// Основные методы
str.length;                    // 13
str.toUpperCase();             // "HELLO, WORLD!"
str.toLowerCase();             // "hello, world!"
str.indexOf('World');         // 7
str.lastIndexOf('o');         // 8
str.slice(7, 12);           // "World"
str.substring(7, 12);        // "World"
str.replace('World', 'JS');  // "Hello, JS!"
str.split(', ');             // ["Hello", "World!"]
str.includes('Hello');        // true
str.startsWith('Hello');      // true
str.endsWith('!');            // true
str.trim();                    // Удалить пробелы с краев

// Шаблонные строки (Template literals)
let name = 'Анна';
let greeting = `Привет, ${name}!
Тебе ${25 + 1} лет.`;

// Конкатенация
'Hello' + ' ' + 'World';  // "Hello World"

7. Set, Map и Date

Set (множество):

let set = new Set([1, 2, 2, 3]); // {1, 2, 3}
set.add(4);
set.delete(1);
set.has(2);                    // true
set.size;                      // 3
set.clear();                   // Очистить

Map (словарь):

let map = new Map();
map.set('name', 'Анна');
map.set(42, 'ответ');
map.get('name');              // 'Анна'
map.has(42);                   // true
map.delete('name');
map.size;                      // 1

Date (дата и время):

let now = new Date();       // Текущая дата
let date = new Date(2024, 0, 1); // 1 января 2024

now.getFullYear();            // 2024
now.getMonth();               // 0-11 (январь = 0)
now.getDate();                // День месяца
now.getDay();                 // День недели (0-6)
now.getHours();               // Часы
now.getMinutes();             // Минуты

now.toLocaleDateString();     // "01.01.2024"
now.toLocaleTimeString();     // "12:30:45"

9. События (Events)

// Добавление обработчика
element.addEventListener('click', function(event) {
    console.log('Клик!', event.target);
});

// Удаление обработчика
function handleClick() {
    console.log('Клик!');
}
element.addEventListener('click', handleClick);
element.removeEventListener('click', handleClick);

// Основные события
'click'      // Клик мышью
'dblclick'   // Двойной клик
'mouseover'  // Наведение мыши
'mouseout'   // Уход мыши
'keydown'    // Нажатие клавиши
'keyup'      // Отпускание клавиши
'submit'     // Отправка формы
'change'     // Изменение значения
'focus'      // Фокус на элементе
'blur'       // Потеря фокуса
'load'       // Загрузка элемента

// Предотвращение действий по умолчанию
form.addEventListener('submit', function(event) {
    event.preventDefault();  // Остановить отправку формы
    // ваша логика
});

10. Консоль и отладка

// Вывод информации
console.log('Обычное сообщение');
console.info('Информация');
console.warn('Предупреждение');
console.error('Ошибка');

// Группировка
console.group('Группа');
console.log('Пункт 1');
console.log('Пункт 2');
console.groupEnd();

// Таблица для объектов/массивов
console.table([
    {name: 'Анна', age: 25},
    {name: 'Иван', age: 30}
]);

// Замер времени
console.time('Таймер');
// Код для замера
console.timeEnd('Таймер');

// Отладка
debugger;  // Точка останова (работает в DevTools)

// Проверка условий
console.assert(x === 5, 'x должен быть равен 5');

// Стек вызовов
console.trace('Трассировка');
